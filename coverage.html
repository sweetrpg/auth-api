package auth0

// Config is the Auth0 tenant this service verifies bearer tokens against.
type Config struct {
        Domain   string
        Audience string
}

// Issuer is the expected `iss` claim for a token from this tenant.
func (c Config) Issuer() string {
        return fmt.Sprintf("https://%s/", c.Domain)
}

// JWKSURL is this tenant's published JWKS endpoint.
func (c Config) JWKSURL() string {
        return fmt.Sprintf("https://%s/.well-known/jwks.json", c.Domain)
}

// ConfigFromEnvironment reads AUTH0_DOMAIN/AUTH0_AUDIENCE. Returns an error
// rather than crashing the process - callers (main.go) are expected to
// treat a missing config as a startup failure, not a mid-request one.
func ConfigFromEnvironment() (Config, error) {
        domain := util.GetEnv(constants.AUTH0_DOMAIN, "")
        audience := util.GetEnv(constants.AUTH0_AUDIENCE, "")
        if domain == "" || audience == "" {
                return Config{}, errors.New("AUTH0_DOMAIN and AUTH0_AUDIENCE must be set in environment")
        }
        return Config{Domain: domain, Audience: audience}, nil
}

// JWKSCache fetches and caches an Auth0 tenant's JWKS. A verification
// attempt against a kid not in the cache triggers exactly one refetch
// (Auth0 rotates signing keys occasionally; a cache built before a rotation
// shouldn't reject a token signed with the new key) before failing as
// invalid. Ports the Swift service's JWKSCache 1:1 - see jwks_cache_test.go
// for the refetch-once behavior this guarantees.
type JWKSCache struct {
        httpClient *http.Client

        // JWKSURLOverride, when set, is fetched instead of config.JWKSURL() - a
        // seam for tests to point the cache at a local stub server rather than a
        // real Auth0 tenant.
        JWKSURLOverride string

        mu sync.RWMutex
        kf keyfunc.Keyfunc
}

// NewJWKSCache returns an empty cache; the first Verify call populates it.
func NewJWKSCache() *JWKSCache {
        return &JWKSCache{httpClient: &http.Client{Timeout: 10 * time.Second}}
}

func (c *JWKSCache) cached() keyfunc.Keyfunc {
        c.mu.RLock()
        defer c.mu.RUnlock()
        return c.kf
}

func (c *JWKSCache) store(kf keyfunc.Keyfunc) {
        c.mu.Lock()
        defer c.mu.Unlock()
        c.kf = kf
}

func (c *JWKSCache) fetch(ctx context.Context, config Config) (keyfunc.Keyfunc, error) {
        url := config.JWKSURL()
        if c.JWKSURLOverride != "" {
                url = c.JWKSURLOverride
        }
        req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
        if err != nil {
                return nil, err
        }
        resp, err := c.httpClient.Do(req)
        if err != nil {
                return nil, fmt.Errorf("failed to fetch Auth0 JWKS: %w", err)
        }
        defer func() { _ = resp.Body.Close() }()
        if resp.StatusCode != http.StatusOK {
                return nil, fmt.Errorf("failed to fetch Auth0 JWKS: status %d", resp.StatusCode)
        }
        body, err := io.ReadAll(resp.Body)
        if err != nil {
                return nil, fmt.Errorf("failed to read Auth0 JWKS response: %w", err)
        }

        kf, err := keyfunc.NewJWKSetJSON(json.RawMessage(body))
        if err != nil {
                return nil, fmt.Errorf("failed to parse Auth0 JWKS: %w", err)
        }
        c.store(kf)
        return kf, nil
}

// Verify verifies token against Auth0's JWKS. It checks the token's kid
// against the current cache *before* parsing (matching the Swift service's
// membership check, not a parse-then-retry loop): a kid already present
// uses the cache with no HTTP call; a kid missing (or no cache fetched yet)
// triggers exactly one fetch, after which the kid is either present or the
// token is rejected - never a second fetch for the same Verify call, since
// a kid genuinely absent from the tenant's JWKS won't appear on a retry.
func (c *JWKSCache) Verify(ctx context.Context, token string, config Config) (*Claims, error) {
        kid, err := extractKID(token)
        if err != nil {
                return nil, err
        }

        kf := c.cached()
        if kf == nil || !c.hasKID(ctx, kf, kid) {
                kf, err = c.fetch(ctx, config)
                if err != nil {
                        return nil, err
                }
        }

        return c.parse(token, kf, config)
}

func (c *JWKSCache) hasKID(ctx context.Context, kf keyfunc.Keyfunc, kid string) bool {
        _, err := kf.Storage().KeyRead(ctx, kid)
        return err == nil
}

func extractKID(token string) (string, error) {
        unverified := &Claims{}
        parsed, _, err := jwt.NewParser().ParseUnverified(token, unverified)
        if err != nil {
                return "", fmt.Errorf("failed to parse token header: %w", err)
        }
        kid, ok := parsed.Header["kid"].(string)
        if !ok || kid == "" {
                return "", jwt.ErrTokenMalformed
        }
        return kid, nil
}

func (c *JWKSCache) parse(token string, kf keyfunc.Keyfunc, config Config) (*Claims, error) {
        claims := &Claims{}
        _, err := jwt.ParseWithClaims(token, claims, kf.Keyfunc,
                jwt.WithValidMethods([]string{"RS256"}),
                jwt.WithIssuer(config.Issuer()),
                jwt.WithAudience(config.Audience),
                jwt.WithExpirationRequired(),
        )
        if err != nil {
                return nil, err
        }
        return claims, nil
}

        database.SetupDatabase()
        defer database.TeardownDatabase()

        setupAcuator(r)

        setupSwagger(r)

        // Per-client/IP rate limiter (Redis-backed, fail-closed). Tier values come from the env

        // chain is not throttled. A rate-limit-store outage returns 503, never unlimited traffic.
        redisPool := apiutil.RedisPool()
        if redisPool != nil {
                pingCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                if err := ratelimit.Ping(pingCtx, redisPool); err != nil {
                        logging.Logger.Error("REDIS_HOST is configured but unreachable at startup; rate-limited requests will fail closed until it recovers",
                                "error", err.Error())
                }
                cancel()
        } else {
                logging.Logger.Warn("REDIS_HOST is not configured; rate limiting will fail closed (503) on every limited request")
        }
        r.Use(ratelimit.Middleware(redisPool, ratelimit.DefaultOptions()))

        if natsURL := os.Getenv("NATS_URL"); natsURL == "" {
                logging.Logger.Warn("NATS_URL is not set; deny and role-removal events are not published")
        } else {
                bus, err := events.ConnectNATS(natsURL)
                if err != nil {
                        log.Fatalf("auth-api cannot start: %v", err)
                }
                defer bus.Close()
                server.SetEventEmitter(events.NewEmitter(bus, events.DefaultBackoff))
        }

        server.SetupHandlers(r, jwksCache, auth0Config)
        fgaStore := setupFGAStore()
        server.SetupFGAHandlers(r, jwksCache, auth0Config, fgaStore)
        sessionLimits := server.SessionLimitsFromEnv()
        var sessionLimiter *ratelimit.Limiter
        if redisPool != nil {
                sessionLimiter = ratelimit.New(redisPool, server.SessionTiers(sessionLimits))
        }
        setupSessionEndpoints(r, jwksCache, auth0Config, fgaStore, sessionLimiter, sessionLimits)

        _ = r.Run(util.GetEnv(apiconstants.BIND_ADDRESS, ":8000"))
}

// setupSessionEndpoints registers the PADR-0040 session endpoints. With no Redis pool the limiter is
// nil and every session request answers 503 rather than run unlimited.
func setupSessionEndpoints(r *gin.Engine, cache *auth0.JWKSCache, cfg auth0.Config, store fga.Store, limiter *ratelimit.Limiter, limits server.SessionLimits) {
        if err := models.EnsureSessionMintIndexes(context.Background()); err != nil {
                logging.Logger.Error("Could not create session mint indexes", "error", err.Error())
                os.Exit(1)
        }
        server.SetupSessionHandlers(r, cache, cfg, store, models.MintStore{}, limiter, limits)
}

// setupFGAStore bootstraps the OpenFGA store and model and returns the client. A misconfigured or
// unreachable OpenFGA fails startup, so the pod restarts rather than serving checks it cannot make.
func setupFGAStore() fga.Store {
        cfg, err := fga.ConfigFromEnvironment()
        if err != nil {
                logging.Logger.Error("OpenFGA configuration invalid", "error", err.Error())
                os.Exit(1)
        }
        boot, err := fga.Bootstrap(context.Background(), cfg.APIURL, cfg.StoreName, cfg.APIKey)
        if err != nil {
                logging.Logger.Error("OpenFGA bootstrap failed", "error", err.Error())
                os.Exit(1)
        }
        client, err := fga.NewClient(cfg.APIURL, boot.StoreID, boot.ModelID, cfg.APIKey)
        if err != nil {
                logging.Logger.Error("OpenFGA client setup failed", "error", err.Error())
                os.Exit(1)
        }
        return client
}

// setupProfiling starts continuous profiling only when the profiling-enabled
// feature flag evaluates to true, regardless of whether
// PYROSCOPE_SERVER_ADDRESS happens to be set - the flag is the on/off
// control, PYROSCOPE_SERVER_ADDRESS is only the destination. See the
// pyroscope-profiling-flag spec's three scenarios.
func setupProfiling(ff *featureflags.Client) func() {
        logging.Logger.Info("Setting up continuous profiling...")

        if !ff.BoolFlag(context.Background(), constants.ProfilingEnabledFlag, false) {
                logging.Logger.Info("profiling-enabled flag is off, continuous profiling disabled")
                return nil
        }

        serverAddress, found := os.LookupEnv(constants.PYROSCOPE_SERVER_ADDRESS)
        if !found {
                logging.Logger.Warn("profiling-enabled flag is on but PYROSCOPE_SERVER_ADDRESS not set, continuous profiling disabled")
                return nil
        }

        profiler, err := pyroscope.Start(pyroscope.Config{
                ApplicationName: constants.ServiceName,
                ServerAddress:   serverAddress,
                TenantID:        util.GetEnv(constants.PYROSCOPE_TENANT_ID, ""),
                Tags: map[string]string{
                        "env": util.GetEnv(apiconstants.ENV, "dev"),
                },
        })
        if err != nil {
                logging.Logger.Error("Error while trying to initialize continuous profiling", "error", err.Error())
                return nil
        }

        return func() {
                _ = profiler.Stop()
        }
}

// Command backfill-audit-fields stamps the platform audit fields (PADR-0001) on auth-api's
// pre-convention documents in user_roles and service_deny_entries: it renames the legacy
// camelCase createdAt to created_at, and sets created_by / updated_by / updated_at. No acting
// user was recorded historically, so *_by is set to the "system" sentinel; updated_at mirrors
// created_at (these rows are only ever added or removed, never updated in place).
//
// user_roles and service_deny_entries are hard-delete security-control records (PADR-0027) -
// there is no deleted_at / deleted_by pair to backfill. The retained "who revoked this" record
// is the admin_action_audit_logs collection, not touched here.
//
// Idempotent - a document that already has created_at is skipped. Dry run by default.
//
//        go run ./cmd/backfill-audit-fields          # report counts, write nothing
//        go run ./cmd/backfill-audit-fields -apply   # perform the writes

}

// stamp backfills every doc in coll whose created_at is missing: created_at from the legacy
// createdAt (or now), updated_at = created_at, created_by / updated_by = "system". Also unsets
// the legacy createdAt key.
func stamp(ctx context.Context, coll string, apply bool) int {
        filter := bson.D{{Key: "created_at", Value: bson.D{{Key: "$exists", Value: false}}}}
        cur, err := database.Db.Collection(coll).Find(ctx, filter)
        if err != nil {
                logging.Logger.Error("query failed", "collection", coll, "error", err.Error())
                return 0
        }
        var docs []bson.Raw
        if err := cur.All(ctx, &docs); err != nil {
                logging.Logger.Error("cursor read failed", "collection", coll, "error", err.Error())
                return 0
        }
        n := 0
        for _, d := range docs {
                createdAt := time.Now().UTC()
                if t, ok := d.Lookup("createdAt").TimeOK(); ok {
                        createdAt = t
                }
                if !apply {
                        n++
                        continue
                }
                _, err := database.Db.Collection(coll).UpdateOne(ctx,
                        bson.D{{Key: "_id", Value: d.Lookup("_id")}},
                        bson.D{
                                {Key: "$set", Value: bson.D{
                                        {Key: "created_at", Value: createdAt},
                                        {Key: "created_by", Value: systemActor},
                                        {Key: "updated_at", Value: createdAt},
                                        {Key: "updated_by", Value: systemActor},
                                }},
                                {Key: "$unset", Value: bson.D{{Key: "createdAt", Value: ""}}},
                        })
                if err != nil {
                        logging.Logger.Error("update failed", "collection", coll, "error", err.Error())
                        continue
                }
                n++
        }
        return n
}

// Command backfill-canonical-user-ids rewrites auth-api's audit *_by fields from raw

// Idempotent: skips values that are already 24-hex UUIDs or "system". Dry run by default.

func backfillCollection(ctx context.Context, coll string, byFields []string, usersBaseURL, internalToken string, apply bool) int {
        // Find docs where any byField is a subject (not a UUID or "system")
        // We'll fetch all docs and filter in Go for simplicity, since these collections are small.
        filter := bson.D{}
        cur, err := database.Db.Collection(coll).Find(ctx, filter)
        if err != nil {
                logging.Logger.Error("query failed", "collection", coll, "error", err.Error())
                return 0
        }
        var docs []bson.Raw
        if err := cur.All(ctx, &docs); err != nil {
                logging.Logger.Error("cursor read failed", "collection", coll, "error", err.Error())
                return 0
        }

        // Collect all distinct subject-shaped values
        subjectSet := make(map[string]struct{})
        for _, d := range docs {
                for _, field := range byFields {
                        if val, ok := d.Lookup(field).StringValueOK(); ok && isSubject(val) {
                                subjectSet[val] = struct{}{}
                        }
                }
        }

        if len(subjectSet) == 0 {
                logging.Logger.Info("no subjects to resolve", "collection", coll)
                return 0
        }

        // Resolve all subjects in one batch
        subjects := make([]string, 0, len(subjectSet))
        for s := range subjectSet {
                subjects = append(subjects, s)
        }
        resolved, err := resolveSubjects(ctx, usersBaseURL, internalToken, subjects)
        if err != nil {
                logging.Logger.Error("resolve failed", "collection", coll, "error", err.Error())
                return 0
        }

        n := 0
        for _, d := range docs {
                id := d.Lookup("_id")
                setFields := bson.D{}
                for _, field := range byFields {
                        if val, ok := d.Lookup(field).StringValueOK(); ok && isSubject(val) {
                                if uid, ok := resolved[val]; ok {
                                        setFields = append(setFields, bson.E{Key: field, Value: uid})
                                } else {
                                        setFields = append(setFields, bson.E{Key: field, Value: systemActor})
                                }
                        }
                }
                if len(setFields) == 0 {
                        continue
                }
                if !apply {
                        n++
                        continue
                }
                _, err := database.Db.Collection(coll).UpdateOne(ctx,
                        bson.D{{Key: "_id", Value: id}},
                        bson.D{{Key: "$set", Value: setFields}},
                )
                if err != nil {
                        logging.Logger.Error("update failed", "collection", coll, "id", id, "error", err.Error())
                        continue
                }
                n++
        }
        return n
}

func backfillAuditLogs(ctx context.Context, usersBaseURL, internalToken string, apply bool) int {

        for _, d := range docs {
                // actingUserSub field in old docs, acting_user_id in new
                if val, ok := d.Lookup("actingUserSub").StringValueOK(); ok && isSubject(val) {

        if err != nil {
                logging.Logger.Error("resolve failed", "collection", constants.AdminActionAuditLogCollection, "error", err.Error())
                return 0
        }

        n := 0
        for _, d := range docs {
                id := d.Lookup("_id")

                if val, ok := d.Lookup("actingUserSub").StringValueOK(); ok && isSubject(val) {
                        if uid, ok := resolved[val]; ok {
                                setFields = append(setFields, bson.E{Key: "acting_user_id", Value: uid})
                        } else {
                                setFields = append(setFields, bson.E{Key: "acting_user_id", Value: systemActor})
                        }
                        // Also unset the old field
                        setFields = append(setFields, bson.E{Key: "$unset", Value: bson.D{{Key: "actingUserSub", Value: ""}}})
                } else if val, ok := d.Lookup("acting_user_id").StringValueOK(); ok && isSubject(val) {
                        if uid, ok := resolved[val]; ok {
                                setFields = append(setFields, bson.E{Key: "acting_user_id", Value: uid})
                        } else {
                                setFields = append(setFields, bson.E{Key: "acting_user_id", Value: systemActor})

                        logging.Logger.Error("update failed", "collection", constants.AdminActionAuditLogCollection, "id", id, "error", err.Error())
                        continue
                }
                n++
        }
        return n
}

func isSubject(val string) bool {
        // Auth0 subjects are not 24-hex and not "system"
        if val == systemActor {
                return false
        }
        if len(val) == 24 {
                // Check if all hex
                for i := 0; i < 24; i++ {
                        c := val[i]
                        if (c < '0' || c > '9') && (c < 'a' || c > 'f') && (c < 'A' || c > 'F') {
                                return true // not all hex, treat as subject
                        }
                }
                return false // 24-hex UUID, already canonical
        }
        return true // treat as subject
}

func resolveSubjects(ctx context.Context, usersBaseURL, internalToken string, subjects []string) (map[string]string, error) {
        type resolveRequest struct {
                Subjects []string `json:"subjects"`
        }
        type resolveResponse map[string]string

        body, err := json.Marshal(resolveRequest{Subjects: subjects})
        if err != nil {
                return nil, fmt.Errorf("marshal request: %w", err)
        }

        httpClient := &http.Client{Timeout: 10 * time.Second}
        req, err := http.NewRequestWithContext(ctx, http.MethodPost, usersBaseURL+"/internal/resolve-subjects", bytes.NewReader(body))
        if err != nil {
                return nil, fmt.Errorf("build request: %w", err)
        }
        req.Header.Set("Content-Type", "application/json")
        req.Header.Set("Authorization", "Bearer "+internalToken)

        resp, err := httpClient.Do(req)
        if err != nil {
                return nil, fmt.Errorf("request failed: %w", err)
        }
        defer func() { _ = resp.Body.Close() }()

        if resp.StatusCode != http.StatusOK {
                return nil, fmt.Errorf("unexpected status %d from users-api", resp.StatusCode)
        }

        var out resolveResponse
        if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
                return nil, fmt.Errorf("decode response: %w", err)
        }
        return out, nil
}

// Command backfill-user-ids sets user_id on user_roles and service_deny_entries rows that were
// written keyed by Auth0 subject only, so roles and deny entries follow the person across linked
// logins. Idempotent: rows that already have a user_id, and subjects users-api does not know, are
// left alone.
//
// Runs as an administrator (PADR-0037): put an admin user's bearer token in ADMIN_BEARER_TOKEN,
// the same token the sweetrpg CLI uses. Needs DB_* (or DB_URI) and USERS_API_URL.
//
//        ADMIN_BEARER_TOKEN=... go run ./cmd/backfill-user-ids
package main

import (
        "context"
        "os"

        "github.com/joho/godotenv"
        "github.com/sweetrpg/auth-api/constants"
        "github.com/sweetrpg/auth-api/identity"
        "github.com/sweetrpg/auth-api/models"
        "github.com/sweetrpg/common.go/logging"
        "github.com/sweetrpg/common.go/util"
        "github.com/sweetrpg/mongodb.go/database"
)

func main() {
        _ = godotenv.Load(".env")
        logging.Init()

        token := util.GetEnv(constants.ADMIN_BEARER_TOKEN, "")
        baseURL := util.GetEnv(constants.USERS_API_URL, "")
        if token == "" || baseURL == "" {
                logging.Logger.Error("ADMIN_BEARER_TOKEN and USERS_API_URL are required")
                os.Exit(1)
        }

        database.SetupDatabase()
        defer database.TeardownDatabase()

        resolver := identity.New(baseURL, nil)
        n, err := models.BackfillUserIDs(context.Background(), func(ctx context.Context, subjects []string) (map[string]string, error) {
                return resolver.Resolve(ctx, subjects, token)
        })
        if err != nil {
                logging.Logger.Error("backfill failed", "error", err.Error())
                os.Exit(1)
        }
        logging.Logger.Info("backfill-user-ids done", "rows_updated", n)
}

// Package events publishes auth-api's deny and role-removal changes to the platform bus
// (PADR-0039) so services holding long-lived credentials for a subject can revoke them. An event
// carries the Auth0 subject and the change kind; it carries no token (PADR-0037).
package events

import (
        "context"
        "encoding/json"
        "fmt"
        "log/slog"
        "strings"
        "sync"
        "time"

        "github.com/google/uuid"
)

// Change kinds, the envelope's action.
const (
        KindDeny        = "deny"
        KindRoleRemoved = "role_removed"
)

// SubjectPrefix starts every event subject: auth.events.<subject token>.
const SubjectPrefix = "auth.events."

// Event is the platform envelope. EntityID is the Auth0 subject. Data.Service names the denied
// service for a deny.
type Event struct {
        EventID    string    `json:"event_id"`
        OccurredAt time.Time `json:"occurred_at"`
        Source     string    `json:"source"`
        EntityType string    `json:"entity_type"`
        EntityID   string    `json:"entity_id"`
        Action     string    `json:"action"`
        Data       EventData `json:"data"`
}

// EventData is the kind-specific detail.
type EventData struct {
        Service string `json:"service,omitempty"`
        Role    string `json:"role,omitempty"`
}

// Publisher puts one message on the bus. msgID lets the bus drop a redelivered duplicate.
type Publisher interface {
        Publish(ctx context.Context, subject, msgID string, body []byte) error
}

// SubjectFor is the event subject for an Auth0 subject. NATS tokens cannot hold '.', '*', '>' or
// whitespace, and Auth0 subjects can (email-based connections), so those become '_'. The real
// subject travels in entity_id.
func SubjectFor(subject string) string {
        return SubjectPrefix + strings.Map(func(r rune) rune {
                switch r {
                case '.', '*', '>', ' ', '\t', '\n', '\r':
                        return '_'
                }
                return r
        }, subject)
}

// Emitter publishes events after the change that caused them committed. A failed publish is
// retried with backoff in the background and never fails the admin request.
type Emitter struct {
        pub     Publisher
        backoff []time.Duration
        wg      sync.WaitGroup
}

// DefaultBackoff is the wait before each retry; after the last the event is dropped and logged.
var DefaultBackoff = []time.Duration{time.Second, 2 * time.Second, 4 * time.Second, 8 * time.Second, 16 * time.Second, 30 * time.Second}

// NewEmitter builds an Emitter. A nil pub drops events with a warning, for deployments with no bus.
func NewEmitter(pub Publisher, backoff []time.Duration) *Emitter {
        return &Emitter{pub: pub, backoff: backoff}
}

// Emit publishes one change for subject without blocking the caller.
func (e *Emitter) Emit(kind, subject string, data EventData) {
        ev := Event{EventID: uuid.NewString(), OccurredAt: time.Now().UTC(), Source: "auth-api", EntityType: "subject", EntityID: subject, Action: kind, Data: data}
        if e == nil || e.pub == nil {
                slog.Warn("auth event not published: no event bus configured", "event_id", ev.EventID, "action", kind)
                return
        }
        body, err := json.Marshal(ev)
        if err != nil {
                slog.Error("auth event not published: marshal failed", "event_id", ev.EventID, "error", err)
                return
        }
        e.wg.Add(1)
        go func() {
                defer e.wg.Done()
                e.send(ev, body)
        }()
}

func (e *Emitter) send(ev Event, body []byte) {
        var err error
        for attempt := 0; ; attempt++ {
                ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                err = e.pub.Publish(ctx, SubjectFor(ev.EntityID), ev.EventID, body)
                cancel()
                if err == nil {
                        return
                }
                if attempt >= len(e.backoff) {
                        break
                }
                slog.Warn("auth event publish failed, retrying", "event_id", ev.EventID, "action", ev.Action, "attempt", attempt+1, "error", err)
                time.Sleep(e.backoff[attempt])
        }
        slog.Error(fmt.Sprintf("auth event dropped after %d retries; the subject's integration credentials are not revoked until reconciled", len(e.backoff)),
                "event_id", ev.EventID, "action", ev.Action, "error", err)
}

// Wait blocks until every in-flight publish has finished or been dropped.
func (e *Emitter) Wait() { e.wg.Wait() }

type NATS struct {
        conn *nats.Conn
        js   jetstream.JetStream
}

// ConnectNATS dials url with NATS_CREDS or NATS_USER/NATS_PASSWORD from the environment.
func ConnectNATS(url string) (*NATS, error) {
        opts := []nats.Option{nats.Name("auth-api"), nats.MaxReconnects(-1)}
        if creds := os.Getenv("NATS_CREDS"); creds != "" {
                opts = append(opts, nats.UserCredentials(creds))
        } else if user := os.Getenv("NATS_USER"); user != "" {
                opts = append(opts, nats.UserInfo(user, os.Getenv("NATS_PASSWORD")))
        }
        conn, err := nats.Connect(url, opts...)
        if err != nil {
                return nil, fmt.Errorf("connect to NATS: %w", err)
        }
        js, err := jetstream.New(conn)
        if err != nil {
                conn.Close()
                return nil, fmt.Errorf("create JetStream context: %w", err)
        }
        return &NATS{conn: conn, js: js}, nil
}

// Publish sends body and waits for the stream's acknowledgement. msgID is the dedupe key.
func (n *NATS) Publish(ctx context.Context, subject, msgID string, body []byte) error {
        _, err := n.js.Publish(ctx, subject, body, jetstream.WithMsgID(msgID))
        return err
}

// Close drains the connection.
func (n *NATS) Close() { _ = n.conn.Drain() }

// Bootstrapped identifies the store and model the service uses.
type Bootstrapped struct {
        StoreID string
        ModelID string
}

// Bootstrap finds or creates the named store, then makes sure its latest authorization model
// matches the checked-in DSL. A model that differs gets written as a new version; an identical
// one is reused, so restarts do not pile up model versions.
func Bootstrap(ctx context.Context, apiURL, storeName, apiKey string) (Bootstrapped, error) {
        admin, err := openfga.NewSdkClient(&openfga.ClientConfiguration{ApiUrl: apiURL, Credentials: apiKeyCredentials(apiKey)})
        if err != nil {
                return Bootstrapped{}, fmt.Errorf("create openfga admin client: %w", err)
        }
        storeID, err := findOrCreateStore(ctx, admin, storeName)
        if err != nil {
                return Bootstrapped{}, err
        }

        scoped, err := openfga.NewSdkClient(&openfga.ClientConfiguration{ApiUrl: apiURL, StoreId: storeID, Credentials: apiKeyCredentials(apiKey)})
        if err != nil {
                return Bootstrapped{}, fmt.Errorf("create openfga store client: %w", err)
        }
        modelID, err := ensureModel(ctx, scoped)
        if err != nil {
                return Bootstrapped{}, err
        }
        logging.Logger.Info("OpenFGA bootstrap complete", "store_id", storeID, "model_id", modelID)
        return Bootstrapped{StoreID: storeID, ModelID: modelID}, nil
}

// findOrCreateStore returns the named store's ID. Replicas starting together can each create one,
// so after creating it re-lists and every replica converges on the oldest store of that name
// (ties by ID); a replica whose own store lost removes it.
func findOrCreateStore(ctx context.Context, admin *openfga.OpenFgaClient, name string) (string, error) {
        existing, err := storesNamed(ctx, admin, name)
        if err != nil {
                return "", err
        }
        if len(existing) > 0 {
                return oldestStore(existing).Id, nil
        }
        created, err := admin.CreateStore(ctx).Body(openfga.ClientCreateStoreRequest{Name: name}).Execute()
        if err != nil {
                return "", fmt.Errorf("create openfga store %q: %w", name, err)
        }
        logging.Logger.Info("Created OpenFGA store", "name", name, "store_id", created.Id)

        all, err := storesNamed(ctx, admin, name)
        if err != nil {
                return "", err
        }
        if len(all) == 0 {
                return created.Id, nil
        }
        winner := oldestStore(all)
        if winner.Id != created.Id {
                if _, err := admin.DeleteStore(ctx).Options(openfga.ClientDeleteStoreOptions{StoreId: &created.Id}).Execute(); err != nil {
                        logging.Logger.Warn("Failed to remove duplicate OpenFGA store", "store_id", created.Id, "error", err.Error())
                }
        }
        return winner.Id, nil
}

func storesNamed(ctx context.Context, admin *openfga.OpenFgaClient, name string) ([]fgasdk.Store, error) {
        var matches []fgasdk.Store
        token := ""
        for {
                opts := openfga.ClientListStoresOptions{}
                if token != "" {
                        opts.ContinuationToken = &token
                }
                page, err := admin.ListStores(ctx).Options(opts).Execute()
                if err != nil {
                        return nil, fmt.Errorf("list openfga stores: %w", err)
                }
                for _, s := range page.Stores {
                        if s.Name == name {
                                matches = append(matches, s)
                        }
                }
                if page.ContinuationToken == "" {
                        return matches, nil
                }
                token = page.ContinuationToken
        }
}

// oldestStore picks the store every replica agrees on. stores must not be empty.
func oldestStore(stores []fgasdk.Store) fgasdk.Store {
        best := stores[0]
        for _, s := range stores[1:] {
                if s.CreatedAt.Before(best.CreatedAt) || (s.CreatedAt.Equal(best.CreatedAt) && s.Id < best.Id) {
                        best = s
                }
        }
        return best
}

func ensureModel(ctx context.Context, scoped *openfga.OpenFgaClient) (string, error) {
        desired, err := desiredModel()
        if err != nil {
                return "", err
        }
        // A store with no model returns no error and a nil model, so any error here is a real failure
        // and must not write a new model version.
        latest, err := scoped.ReadLatestAuthorizationModel(ctx).Execute()
        if err != nil {
                return "", fmt.Errorf("read latest openfga authorization model: %w", err)
        }
        if latest.AuthorizationModel != nil && sameDefinitions(latest.AuthorizationModel.TypeDefinitions, desired.TypeDefinitions) {
                return latest.AuthorizationModel.Id, nil
        }
        written, err := scoped.WriteAuthorizationModel(ctx).Body(desired).Execute()
        if err != nil {
                return "", fmt.Errorf("write openfga authorization model: %w", err)
        }
        return written.AuthorizationModelId, nil
}

import (

// Write failures a caller can act on. Anything else is a transport or server failure.
var (
        // ErrInvalid means OpenFGA rejected the batch: an unknown relation, a tuple that violates the
        // model, or a malformed key. Retrying the same batch cannot succeed.
        ErrInvalid = errors.New("openfga rejected the tuple batch")
        // ErrConflict means the batch conflicts with stored state (HTTP 409).
        ErrConflict = errors.New("openfga tuple batch conflicts with stored state")
)

// Tuple is one relationship: User has Relation on Object, e.g. user:alice approver volume-edit:1234.
type Tuple struct {
        User     string `json:"user"`
        Relation string `json:"relation"`
        Object   string `json:"object"`
}

// Store is the subset of OpenFGA the platform uses. Handlers depend on this interface so tests
// run without a server.
type Store interface {
        Check(ctx context.Context, t Tuple) (bool, error)
        Write(ctx context.Context, writes, deletes []Tuple) error
        Read(ctx context.Context, object string, pageSize int32, continuation string) ([]Tuple, string, error)
}

// Client is the OpenFGA-backed Store, bound to one store and authorization model.
type Client struct {
        sdk *openfga.OpenFgaClient
}

// NewClient builds a Client for a store and model that Bootstrap resolved.
// apiKey is the OpenFGA preshared key, or empty when authn is off.
func NewClient(apiURL, storeID, modelID, apiKey string) (*Client, error) {
        sdk, err := openfga.NewSdkClient(&openfga.ClientConfiguration{
                ApiUrl:               apiURL,
                StoreId:              storeID,
                AuthorizationModelId: modelID,
                Credentials:          apiKeyCredentials(apiKey),
        })
        if err != nil {
                return nil, fmt.Errorf("create openfga client: %w", err)
        }
        return &Client{sdk: sdk}, nil
}

// apiKeyCredentials sends key as a bearer token on every OpenFGA call, or nothing when it is empty.
func apiKeyCredentials(key string) *credentials.Credentials {
        if key == "" {
                return nil
        }
        return &credentials.Credentials{
                Method: credentials.CredentialsMethodApiToken,
                Config: &credentials.Config{ApiToken: key},
        }
}

// Check reports whether t.User holds t.Relation on t.Object.
func (c *Client) Check(ctx context.Context, t Tuple) (bool, error) {
        resp, err := c.sdk.Check(ctx).Body(openfga.ClientCheckRequest{
                User:     t.User,
                Relation: t.Relation,
                Object:   t.Object,
        }).Execute()
        if err != nil {
                return false, fmt.Errorf("openfga check: %w", err)
        }
        return resp.GetAllowed(), nil
}

// Write applies writes and deletes in one OpenFGA request. The SDK sends it as a single
// transaction by default, so a failure applies none of them. Writing a tuple that already exists
// and deleting one that does not are ignored, so a reconciliation can resend a batch safely.
// A rejected batch returns ErrInvalid or ErrConflict; the returned text carries no OpenFGA
// response detail because the caller stores it in the audit log.
func (c *Client) Write(ctx context.Context, writes, deletes []Tuple) error {
        body := openfga.ClientWriteRequest{
                Writes:  toClientKeys(writes),
                Deletes: toClientDeletes(deletes),
        }
        opts := openfga.ClientWriteOptions{Conflict: openfga.ClientWriteConflictOptions{
                OnDuplicateWrites: openfga.CLIENT_WRITE_REQUEST_ON_DUPLICATE_WRITES_IGNORE,
                OnMissingDeletes:  openfga.CLIENT_WRITE_REQUEST_ON_MISSING_DELETES_IGNORE,
        }}
        if _, err := c.sdk.Write(ctx).Body(body).Options(opts).Execute(); err != nil {
                logging.Logger.Error("OpenFGA write failed", "error", err.Error())
                return classifyWriteError(err)
        }
        return nil
}

func classifyWriteError(err error) error {
        var validation fgasdk.FgaApiValidationError
        if errors.As(err, &validation) {
                return fmt.Errorf("%w: %s", ErrInvalid, validation.ResponseCode())
        }
        var api fgasdk.FgaApiError
        if errors.As(err, &api) && api.ResponseStatusCode() == http.StatusConflict {
                return ErrConflict
        }
        return errors.New("openfga write failed")
}

// Read returns one page of tuples on object and the continuation token for the next page, empty
// when there are no more.
func (c *Client) Read(ctx context.Context, object string, pageSize int32, continuation string) ([]Tuple, string, error) {
        opts := openfga.ClientReadOptions{PageSize: &pageSize}
        if continuation != "" {
                opts.ContinuationToken = &continuation
        }
        resp, err := c.sdk.Read(ctx).Body(openfga.ClientReadRequest{Object: &object}).Options(opts).Execute()
        if err != nil {
                return nil, "", fmt.Errorf("openfga read: %w", err)
        }
        tuples := make([]Tuple, 0, len(resp.Tuples))
        for _, t := range resp.Tuples {
                key := t.GetKey()
                tuples = append(tuples, Tuple{User: key.GetUser(), Relation: key.GetRelation(), Object: key.GetObject()})
        }
        return tuples, resp.GetContinuationToken(), nil
}

func toClientKeys(ts []Tuple) []openfga.ClientTupleKey {
        if len(ts) == 0 {
                return nil
        }
        keys := make([]openfga.ClientTupleKey, 0, len(ts))
        for _, t := range ts {
                keys = append(keys, openfga.ClientTupleKey{User: t.User, Relation: t.Relation, Object: t.Object})
        }
        return keys
}

func toClientDeletes(ts []Tuple) []openfga.ClientTupleKeyWithoutCondition {
        if len(ts) == 0 {
                return nil
        }
        keys := make([]openfga.ClientTupleKeyWithoutCondition, 0, len(ts))
        for _, t := range ts {
                keys = append(keys, openfga.ClientTupleKeyWithoutCondition{User: t.User, Relation: t.Relation, Object: t.Object})
        }
        return keys
}

        }

}

// sameDefinitions reports whether two sets of type definitions encode identically. The SDK
// serializes both sides the same way, so byte equality matches semantic equality here.
func sameDefinitions(a, b []fgasdk.TypeDefinition) bool {
        ja, errA := json.Marshal(a)
        jb, errB := json.Marshal(b)
        return errA == nil && errB == nil && bytes.Equal(ja, jb)
}

// Package identity maps an Auth0 subject to the canonical user it belongs to, so roles and
// deny entries follow the person across their linked logins. Every lookup uses a real user's
// bearer token (PADR-0037): the caller's own for their own subject, an administrator's for
// someone else's. There is no service credential.
package identity

import (
        "bytes"
        "context"
        "encoding/json"
        "fmt"
        "net/http"
        "sync"
        "time"

        "github.com/sweetrpg/common.go/logging"
)

const (
        foundTTL    = time.Minute
        notFoundTTL = 10 * time.Second
        maxBatch    = 200
)

// OwnLookup resolves the canonical user ID of the bearer token's own user (users-api GET
// /profile); "" when it cannot.
type OwnLookup func(ctx context.Context, token string) string

// Resolver caches subject -> canonical user answers briefly because /authz/check asks on every
// request. A failed or impossible lookup reports "" so callers fall back to the subject alone.
type Resolver struct {
        baseURL string
        own     OwnLookup
        http    *http.Client
        now     func() time.Time

        mu    sync.Mutex
        cache map[string]entry
}

type entry struct {
        userID  string
        expires time.Time
}

// New returns a Resolver. baseURL is users-api's; own looks up a caller's own user.
func New(baseURL string, own OwnLookup) *Resolver {
        return &Resolver{baseURL: baseURL, own: own, http: &http.Client{Timeout: 3 * time.Second}, now: time.Now, cache: map[string]entry{}}
}

// Cached returns a fresh cached answer for subject without any network call.
func (r *Resolver) Cached(subject string) string {
        if r == nil {
                return ""
        }
        r.mu.Lock()
        defer r.mu.Unlock()
        if e, ok := r.cache[subject]; ok && r.now().Before(e.expires) {
                return e.userID
        }
        return ""
}

// Own returns the canonical user of the caller whose verified subject is subject, looked up with
// the caller's own bearer token. "" when the token is missing or the lookup fails.
func (r *Resolver) Own(ctx context.Context, subject, token string) string {
        if r == nil || subject == "" || token == "" {
                return ""
        }
        if e, ok := r.entry(subject); ok {
                return e.userID
        }
        id := ""
        if r.own != nil {
                id = r.own(ctx, token)
        }
        r.store(subject, id)
        return id
}

// Other returns the canonical user of someone else's subject, looked up with an administrator's
// bearer token. "" when there is no admin token (a caller using another credential) or the
// lookup fails, with a WARN so the gap is visible.
func (r *Resolver) Other(ctx context.Context, subject, adminToken string) string {
        if r == nil || subject == "" {
                return ""
        }
        if e, ok := r.entry(subject); ok {
                return e.userID
        }
        if adminToken == "" {
                logging.Logger.Warn("no admin bearer token to resolve the subject's canonical user; using the subject alone", "subject", subject)
                return ""
        }
        resolved, err := r.Resolve(ctx, []string{subject}, adminToken)
        if err != nil {
                logging.Logger.Warn("could not resolve the subject's canonical user; using the subject alone", "subject", subject, "error", err.Error())
                return ""
        }
        r.store(subject, resolved[subject])
        return resolved[subject]
}

// Resolve maps subjects to canonical user IDs through users-api's POST /internal/resolve-subjects,
// authorized by an administrator's bearer token. Subjects with no user are absent from the result.
func (r *Resolver) Resolve(ctx context.Context, subjects []string, adminToken string) (map[string]string, error) {
        if r == nil || r.baseURL == "" {
                return nil, fmt.Errorf("USERS_API_URL is not configured")
        }
        if adminToken == "" {
                return nil, fmt.Errorf("an administrator bearer token is required")
        }
        out := map[string]string{}
        for start := 0; start < len(subjects); start += maxBatch {
                batch := subjects[start:min(start+maxBatch, len(subjects))]
                body, err := json.Marshal(map[string][]string{"subjects": batch})
                if err != nil {
                        return nil, err
                }
                req, err := http.NewRequestWithContext(ctx, http.MethodPost, r.baseURL+"/internal/resolve-subjects", bytes.NewReader(body))
                if err != nil {
                        return nil, err
                }
                req.Header.Set("Content-Type", "application/json")
                req.Header.Set("Authorization", "Bearer "+adminToken)
                resp, err := r.http.Do(req)
                if err != nil {
                        return nil, err
                }
                var part map[string]string
                if err := decode(resp, &part); err != nil {
                        return nil, err
                }
                for k, v := range part {
                        out[k] = v
                }
        }
        return out, nil
}

func decode(resp *http.Response, v any) error {
        defer func() { _ = resp.Body.Close() }()
        if resp.StatusCode != http.StatusOK {
                return fmt.Errorf("users-api answered %d", resp.StatusCode)
        }
        return json.NewDecoder(resp.Body).Decode(v)
}

func (r *Resolver) entry(subject string) (entry, bool) {
        r.mu.Lock()
        defer r.mu.Unlock()
        e, ok := r.cache[subject]
        if !ok || !r.now().Before(e.expires) {
                return entry{}, false
        }
        return e, true
}

func (r *Resolver) store(subject, userID string) {
        ttl := foundTTL
        if userID == "" {
                ttl = notFoundTTL
        }
        r.mu.Lock()
        defer r.mu.Unlock()
        r.cache[subject] = entry{userID: userID, expires: r.now().Add(ttl)}
}

// AuditStatus tracks whether the mutation an AdminActionAuditLog record
// covers actually completed.
type AuditStatus string

const (
        AuditAttempted AuditStatus = "attempted"
        AuditSucceeded AuditStatus = "succeeded"
        AuditFailed    AuditStatus = "failed"
)

// AdminActionAuditLog attributes a write-route mutation (grant/revoke a
// role, add/remove a service deny entry) to the acting user who made it.
// Written *before* the mutation is attempted (status "attempted") and
// updated *after* it completes - see RecordAuditAttempt and CompleteAudit.
// If the "before" write fails, the mutation must not be performed: an audit
// trail that can silently fail to exist is not an audit trail.
// ActingUserID is the canonical users._id of the acting admin (resolved from
// their Auth0 subject at the request boundary). TargetSubject remains the
// raw Auth0 sub of the user being modified - that is the identity being
// authorized, not the acting user.
type AdminActionAuditLog struct {
        ID            uuid.UUID   `bson:"_id" json:"id"`
        ActingUserID  string      `bson:"acting_user_id" json:"acting_user_id"`
        Action        string      `bson:"action" json:"action"`
        TargetSubject string      `bson:"targetSubject" json:"targetSubject"`
        Detail        string      `bson:"detail" json:"detail"`
        Status        AuditStatus `bson:"status" json:"status"`
        AttemptedAt   time.Time   `bson:"attemptedAt" json:"attemptedAt"`
        CompletedAt   *time.Time  `bson:"completedAt,omitempty" json:"completedAt,omitempty"`
        ErrorMessage  string      `bson:"errorMessage,omitempty" json:"errorMessage,omitempty"`
}

// RecordAuditAttempt writes the "attempted" audit record for a write-route
// mutation before the mutation itself runs. Returns the record's ID for a
// later CompleteAudit call. Callers must not perform the mutation if this
// returns an error.
func RecordAuditAttempt(ctx context.Context, actingUserID, action, targetSubject, detail string) (uuid.UUID, error) {
        entry := AdminActionAuditLog{
                ID:            uuid.New(),
                ActingUserID:  actingUserID,
                Action:        action,
                TargetSubject: targetSubject,
                Detail:        detail,
                Status:        AuditAttempted,
                AttemptedAt:   time.Now().UTC(),
        }
        _, err := database.Db.Collection(constants.AdminActionAuditLogCollection).InsertOne(ctx, entry)
        return entry.ID, err
}

// CompleteAudit updates an audit record to its final status after the
// mutation it covers has run. Best-effort: a failure here doesn't undo a
// mutation that already happened, so callers should log a warning rather
// than fail the request on error.
func CompleteAudit(ctx context.Context, id uuid.UUID, status AuditStatus, errMessage string) error {
        update := bson.M{"status": status, "completedAt": time.Now().UTC()}
        if errMessage != "" {
                update["errorMessage"] = errMessage
        }
        _, err := database.Db.Collection(constants.AdminActionAuditLogCollection).UpdateByID(ctx, id, bson.M{"$set": update})
        return err
}

// SubjectResolver maps Auth0 subjects to canonical user IDs; unknown subjects are omitted.
type SubjectResolver func(ctx context.Context, subjects []string) (map[string]string, error)

// BackfillUserIDs sets user_id on role and deny-entry rows written when they were keyed by login
// only, so they follow the person. Idempotent: rows that have a user_id, and subjects users-api
// does not know, are left alone. It returns how many rows it updated.
func BackfillUserIDs(ctx context.Context, resolve SubjectResolver) (int64, error) {
        var total int64
        for _, coll := range []string{constants.UserRolesCollection, constants.ServiceDenyEntriesCollection} {
                n, err := backfillCollection(ctx, coll, resolve)
                total += n
                if err != nil {
                        return total, fmt.Errorf("backfilling %s: %w", coll, err)
                }
        }
        return total, nil
}

func backfillCollection(ctx context.Context, name string, resolve SubjectResolver) (int64, error) {
        coll := database.Db.Collection(name)
        missing := bson.D{{Key: "user_id", Value: bson.D{{Key: "$exists", Value: false}}}}
        values, err := coll.Distinct(ctx, "subject", missing)
        if err != nil {
                return 0, err
        }
        subjects := make([]string, 0, len(values))
        for _, v := range values {
                if s, ok := v.(string); ok && s != "" {
                        subjects = append(subjects, s)
                }
        }
        if len(subjects) == 0 {
                return 0, nil
        }
        resolved, err := resolve(ctx, subjects)
        if err != nil {
                return 0, err
        }
        var updated int64
        for subject, userID := range resolved {
                if userID == "" {
                        continue
                }
                res, err := coll.UpdateMany(ctx,
                        bson.D{{Key: "subject", Value: subject}, {Key: "user_id", Value: bson.D{{Key: "$exists", Value: false}}}},
                        bson.D{{Key: "$set", Value: bson.D{{Key: "user_id", Value: userID}}}})
                if err != nil {
                        return updated, err
                }
                updated += res.ModifiedCount
        }
        return updated, nil
}

// Principal is who a role or deny entry applies to: a login (Auth0 subject) and, when known, the
// canonical user that login belongs to. Rows are matched by either, so a role granted to the
// person reaches every linked login, and a row written before user IDs existed still matches
// through its subject.
type Principal struct {
        Subject string
        UserID  string
}

// filter matches the principal's rows by user_id or subject.
func (p Principal) filter() bson.D {
        if p.UserID == "" {
                return bson.D{{Key: "subject", Value: p.Subject}}
        }
        return bson.D{{Key: "$or", Value: bson.A{
                bson.D{{Key: "user_id", Value: p.UserID}},
                bson.D{{Key: "subject", Value: p.Subject}},
        }}}
}

// with narrows the principal filter by one more field.
func (p Principal) with(key string, value any) bson.D {
        return bson.D{{Key: "$and", Value: bson.A{p.filter(), bson.D{{Key: key, Value: value}}}}}
}

// ClosedMintRetention is how long a closed mint record is kept before the TTL index removes it. The
// audit log keeps the history (PADR-0040).
const ClosedMintRetention = 30 * 24 * time.Hour

// SessionMint records that auth-api minted a session ID for a user and wrote the GM tuple. It is
// open until the session is ended or garbage collected. ClosedAt is absent while open, which the
// TTL index relies on: only a closed record carries the date.
type SessionMint struct {
        ID        string     `bson:"_id" json:"id"`
        UserID    string     `bson:"user_id" json:"user_id"`
        CreatedAt time.Time  `bson:"created_at" json:"created_at"`
        CreatedBy string     `bson:"created_by" json:"-"`
        UpdatedAt time.Time  `bson:"updated_at" json:"-"`
        UpdatedBy string     `bson:"updated_by" json:"-"`
        ClosedAt  *time.Time `bson:"closed_at,omitempty" json:"-"`
}

// MintStore is the Mongo-backed store of session mints.
type MintStore struct{}

func mintColl() *mongo.Collection {
        return database.Db.Collection(constants.SessionMintsCollection)
}

// EnsureSessionMintIndexes creates the indexes the mint queries and the retention rule need.
func EnsureSessionMintIndexes(ctx context.Context) error {
        _, err := mintColl().Indexes().CreateMany(ctx, []mongo.IndexModel{
                {Keys: bson.D{{Key: "user_id", Value: 1}, {Key: "closed_at", Value: 1}}},
                {Keys: bson.D{{Key: "closed_at", Value: 1}, {Key: "created_at", Value: 1}, {Key: "_id", Value: 1}}},
                {
                        Keys:    bson.D{{Key: "closed_at", Value: 1}},
                        Options: options.Index().SetExpireAfterSeconds(int32(ClosedMintRetention / time.Second)).SetName("closed_at_ttl"),
                },
        })
        return err
}

// Insert stores a new open mint.
func (MintStore) Insert(ctx context.Context, m SessionMint) error {
        _, err := mintColl().InsertOne(ctx, m)
        return err
}

// CountOpen returns how many open mints userID holds.
func (MintStore) CountOpen(ctx context.Context, userID string) (int64, error) {
        return mintColl().CountDocuments(ctx, bson.D{{Key: "user_id", Value: userID}, {Key: "closed_at", Value: nil}})
}

// Close ends an open mint. It reports whether this call closed it: the match requires closed_at to
// be absent, so closing twice (or racing a garbage collection) is a no-op for the second caller.
func (MintStore) Close(ctx context.Context, id, actor string, now time.Time) (bool, error) {
        res := mintColl().FindOneAndUpdate(ctx,
                bson.D{{Key: "_id", Value: id}, {Key: "closed_at", Value: nil}},
                bson.D{{Key: "$set", Value: bson.D{{Key: "closed_at", Value: now}, {Key: "updated_at", Value: now}, {Key: "updated_by", Value: actor}}}})
        if err := res.Err(); err != nil {
                if errors.Is(err, mongo.ErrNoDocuments) {
                        return false, nil
                }
                return false, err
        }
        return true, nil
}

// MintCursor is the position after which ListOpenBefore continues: the last mint returned.
type MintCursor struct {
        CreatedAt time.Time
        ID        string
}

// ListOpenBefore returns open mints created at or before cutoff, ordered by (created_at, _id),
// starting after cursor when it is non-nil, at most limit. Paging by cursor means a long backlog
// of old mints cannot hide a newer one.
func (MintStore) ListOpenBefore(ctx context.Context, cutoff time.Time, after *MintCursor, limit int64) ([]SessionMint, error) {
        filter := bson.D{{Key: "closed_at", Value: nil}, {Key: "created_at", Value: bson.D{{Key: "$lte", Value: cutoff}}}}
        if after != nil {
                filter = bson.D{
                        {Key: "closed_at", Value: nil},
                        {Key: "$and", Value: bson.A{
                                bson.D{{Key: "created_at", Value: bson.D{{Key: "$lte", Value: cutoff}}}},
                                bson.D{{Key: "$or", Value: bson.A{
                                        bson.D{{Key: "created_at", Value: bson.D{{Key: "$gt", Value: after.CreatedAt}}}},
                                        bson.D{{Key: "created_at", Value: after.CreatedAt}, {Key: "_id", Value: bson.D{{Key: "$gt", Value: after.ID}}}},
                                }}},
                        }},
                }
        }
        cur, err := mintColl().Find(ctx, filter,
                options.Find().SetSort(bson.D{{Key: "created_at", Value: 1}, {Key: "_id", Value: 1}}).SetLimit(limit))
        if err != nil {
                return nil, err
        }
        var out []SessionMint
        if err := cur.All(ctx, &out); err != nil {
                return nil, err
        }
        return out, nil
}

// FindOpen returns the open mint with id, and whether there is one.
func (MintStore) FindOpen(ctx context.Context, id string) (SessionMint, bool, error) {
        var m SessionMint
        err := mintColl().FindOne(ctx, bson.D{{Key: "_id", Value: id}, {Key: "closed_at", Value: nil}}).Decode(&m)
        if errors.Is(err, mongo.ErrNoDocuments) {
                return m, false, nil
        }
        return m, err == nil, err
}

func setupAuthzHandlers(g *gin.Engine, cache *auth0.JWKSCache, config auth0.Config) {

// Check whether a subject may access a service.
//
//         Verifies the bearer token against Auth0's JWKS, then looks up the subject's roles and any deny entry for the requested service.
//                @Summary                Authorize a service request
//                @Description        Authorize a service request
//                @Tags                        authz
//                @Accept                        json
//                @Produce                json
//                @Param                        Authorization        header                string                                true        "Bearer <Auth0 access token>"
//                @Param                        request                        body                authzCheckRequest        true        "Service to check access for"
//                @Success                200                                {object}        authzAllowedResponse
//                @Success                200                                {object}        authzDeniedResponse
//                @Failure                401                                {object}        authzErrorResponse
//                @Router                        /authz/check [post]
func checkAuthz(c *gin.Context, cache *auth0.JWKSCache, config auth0.Config) {
        var req authzCheckRequest
        if err := c.ShouldBindJSON(&req); err != nil {
                invalidTokenResponse(c)
                return
        }

        token := bearerToken(c)
        if token == "" {
                invalidTokenResponse(c)
                return
        }

        claims, err := cache.Verify(c.Request.Context(), token, config)
        if err != nil {
                // Logged at Warn, not Error: an invalid/expired token is a routine client condition,
                // not a service fault. Still logs the real reason (bad signature, expired, wrong
                // issuer/audience, unresolvable kid) so it stays distinguishable from a downstream
                // dependency failure below - the Swift service's equivalent check collapsed both into
                // one generic "invalid_token" response with no logged cause, which cost real debugging
                // time during a MongoDB decode failure that looked identical to an invalid token.
                logging.Logger.Warn("Token verification failed", "error", err.Error())
                invalidTokenResponse(c)
                return
        }

        subject := claims.Subject
        p := cachedPrincipal(subject)
        if req.Service != constants.USERS_API_SERVICE {
                p = ownPrincipal(c.Request.Context(), subject, token)
        }

        denied, err := models.HasDenyEntry(c.Request.Context(), p, req.Service)
        if err != nil {
                logging.Logger.Error("Failed to look up deny entry", "error", err.Error())
                c.JSON(http.StatusInternalServerError, authzErrorResponse{Error: "query_failed"})
                return
        }
        if denied {
                c.JSON(http.StatusOK, authzDeniedResponse{Allowed: false, Reason: "service_denied"})
                return
        }

        roles, err := models.ListRolesFor(c.Request.Context(), p)
        if err != nil {
                logging.Logger.Error("Failed to look up roles", "error", err.Error())
                c.JSON(http.StatusInternalServerError, authzErrorResponse{Error: "query_failed"})
                return
        }

        roleNames := make([]string, 0, len(roles))
        for _, r := range roles {
                roleNames = append(roleNames, string(r.Role))
        }
        if len(roleNames) == 0 {
                roleNames = []string{string(models.RoleUser)}
        }

        c.JSON(http.StatusOK, authzAllowedResponse{Allowed: true, Roles: roleNames, Sub: subject})
}

func invalidTokenResponse(c *gin.Context) {
        c.JSON(http.StatusUnauthorized, authzErrorResponse{Error: "invalid_token"})
}

func bearerToken(c *gin.Context) string {
        const prefix = "Bearer "
        auth := c.GetHeader("Authorization")
        if len(auth) <= len(prefix) || auth[:len(prefix)] != prefix {
                return ""
        }
        return auth[len(prefix):]
}

        store  fga.Store
        cache  *auth0.JWKSCache

// Seams for tests. principalFor resolves the caller's canonical user; holdsWriterRole checks the
// openfga-writer role; auditedWrite records and runs an admin write.
var (
        principalFor    = ownPrincipal
        holdsWriterRole = func(ctx context.Context, p models.Principal) (bool, error) {
                return models.HasRole(ctx, p, models.RoleOpenFGAWriter)
        }
        auditedWrite = performAudited

// SetupFGAHandlers registers POST /authz/check-relation and the internal tuple endpoints. Call
// it once at startup with the store Bootstrap resolved.
func SetupFGAHandlers(g *gin.Engine, cache *auth0.JWKSCache, config auth0.Config, store fga.Store) {
        ensureUserResolver()
        h := fgaHandlers{store: store, cache: cache, config: config}
        g.POST("/authz/check-relation", h.checkRelation)
        g.POST("/internal/openfga/tuples", h.writeTuples)
        g.GET("/internal/openfga/tuples", h.readTuples)
}

// checkRelation answers whether the caller's subject holds a relation on an object. An invalid
// token never reaches the store.
func (h fgaHandlers) checkRelation(c *gin.Context) {
        var req checkRelationRequest
        if err := c.ShouldBindJSON(&req); err != nil || !validObject(req.Object) {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        token := bearerToken(c)
        if token == "" {
                invalidTokenResponse(c)
                return
        }
        subject, err := verifyToken(c.Request.Context(), h.cache, token, h.config)
        if err != nil {
                invalidTokenResponse(c)
                return
        }
        // Tuples belong to the person, so the check is keyed on the canonical user ID. Without one
        // (users-api unreachable, or the login has no user) the answer would depend on which login
        // was used, so fail closed.
        userID := principalFor(c.Request.Context(), subject, token).UserID
        if userID == "" {
                logging.Logger.Warn("OpenFGA check denied; canonical user unresolved", "subject", subject)
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }

        allowed, err := h.store.Check(c.Request.Context(), fga.Tuple{
                User: userPrincipalType + userID, Relation: req.Relation, Object: req.Object,
        })
        if err != nil {
                logging.Logger.Error("OpenFGA check failed; denying", "object", req.Object, "relation", req.Relation, "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        c.JSON(http.StatusOK, checkRelationResponse{Allowed: allowed})
}

// writeTuples applies a batch of tuple writes and deletes atomically, after recording the attempt
// in the audit log.
func (h fgaHandlers) writeTuples(c *gin.Context) {
        subject, ok := h.requireWriter(c)
        if !ok {
                return
        }
        var req tupleWriteRequest
        if err := c.ShouldBindJSON(&req); err != nil || !validTupleBatch(req) {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }

        detail := auditDetail(req)
        _, err := auditedWrite(c.Request.Context(), subject, fgaWriteAction, "", detail, func(ctx context.Context) (int, error) {
                return http.StatusOK, h.store.Write(ctx, req.Writes, req.Deletes)
        })
        if err != nil {
                logging.Logger.Error("OpenFGA tuple write failed", "error", err.Error())
                switch {
                case errors.Is(err, fga.ErrInvalid):
                        c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_tuples"})
                case errors.Is(err, fga.ErrConflict):
                        c.JSON(http.StatusConflict, authzErrorResponse{Error: "tuple_conflict"})
                default:
                        c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                }
                return
        }
        c.JSON(http.StatusOK, gin.H{"writes": len(req.Writes), "deletes": len(req.Deletes)})
}

// readTuples returns one page of tuples on an object for reconciliation.
func (h fgaHandlers) readTuples(c *gin.Context) {
        if _, ok := h.requireWriter(c); !ok {
                return
        }
        object := c.Query("object")
        if !validObject(object) {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        pageSize, ok := parsePageSize(c.Query("page_size"))
        if !ok {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }

        tuples, next, err := h.store.Read(c.Request.Context(), object, pageSize, c.Query("continuation_token"))
        if err != nil {
                logging.Logger.Error("OpenFGA tuple read failed", "object", object, "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        c.JSON(http.StatusOK, tupleReadResponse{Tuples: tuples, ContinuationToken: next})
}

// requireWriter verifies the bearer token and that its subject holds openfga-writer. It writes
// the error response and returns ok=false on any failure.
func (h fgaHandlers) requireWriter(c *gin.Context) (string, bool) {
        return h.requireWriterStatus(c, http.StatusInternalServerError, "query_failed")
}

// requireWriterStatus is requireWriter with the response used when the role lookup itself fails.
// The session-mint endpoints use 503 so a caller retries, as the other fail-closed paths do.
func (h fgaHandlers) requireWriterStatus(c *gin.Context, lookupFailStatus int, lookupFailError string) (string, bool) {
        token := bearerToken(c)
        if token == "" {
                invalidTokenResponse(c)
                return "", false
        }
        subject, err := verifyToken(c.Request.Context(), h.cache, token, h.config)
        if err != nil {
                invalidTokenResponse(c)
                return "", false
        }
        holds, err := holdsWriterRole(c.Request.Context(), principalFor(c.Request.Context(), subject, token))
        if err != nil {
                logging.Logger.Error("Failed to look up roles", "error", err.Error())
                c.JSON(lookupFailStatus, authzErrorResponse{Error: lookupFailError})
                return "", false
        }
        if !holds {
                c.JSON(http.StatusForbidden, authzErrorResponse{Error: "forbidden"})
                return "", false
        }
        return subject, true
}

// validObject requires the "type:id" form OpenFGA uses.
func validObject(s string) bool {
        typ, id, ok := strings.Cut(s, ":")
        return ok && typ != "" && id != "" && !strings.ContainsAny(s, " \t\n#")
}

// auditDetail lists every tuple in the batch (at most maxTupleBatch), so the audit log shows which
// relationships changed.
func auditDetail(req tupleWriteRequest) string {
        var b strings.Builder
        b.WriteString("writes=" + strconv.Itoa(len(req.Writes)) + " deletes=" + strconv.Itoa(len(req.Deletes)))
        for _, t := range req.Writes {
                b.WriteString("; +" + t.User + " " + t.Relation + " " + t.Object)
        }
        for _, t := range req.Deletes {
                b.WriteString("; -" + t.User + " " + t.Relation + " " + t.Object)
        }
        return b.String()
}

// validUser requires "user:<canonical id>". An Auth0 subject ("provider|id") is rejected so
// tuples can't be keyed on a login.
func validUser(s string) bool {
        id, ok := strings.CutPrefix(s, userPrincipalType)
        return ok && id != "" && !strings.ContainsAny(id, " \t\n#|:*")
}

func validTupleBatch(req tupleWriteRequest) bool {
        n := len(req.Writes) + len(req.Deletes)
        if n == 0 || n > maxTupleBatch {
                return false
        }
        for _, t := range append(append([]fga.Tuple{}, req.Writes...), req.Deletes...) {
                if !validUser(t.User) || t.Relation == "" || !validObject(t.Object) {
                        return false
                }
        }
        return true
}

func parsePageSize(raw string) (int32, bool) {
        if raw == "" {
                return defaultTuplePage, true
        }
        n, err := strconv.Atoi(raw)
        if err != nil || n < 1 || n > maxTuplePage {
                return 0, false
        }
        return int32(n), true
}

// userResolver maps a login to its canonical user, so roles and deny entries follow the person
// across linked logins. Nil resolves nothing (subject-only matching).
var userResolver *identity.Resolver

// ensureUserResolver builds the resolver once. Lookups use user bearer tokens only (PADR-0037).
func ensureUserResolver() {
        if userResolver != nil {
                return
        }
        usersBaseURL := util.GetEnv(constants.USERS_API_URL, "")
        userResolver = identity.New(usersBaseURL, authz.NewClient("", usersBaseURL).ResolveUserID)
}

// ownPrincipal is the caller's own login plus the canonical user their own bearer token resolves
// to (users-api GET /profile, cached). token may be empty, which leaves the subject alone.
func ownPrincipal(ctx context.Context, subject, token string) models.Principal {
        return models.Principal{Subject: subject, UserID: userResolver.Own(ctx, subject, token)}
}

// cachedPrincipal uses only an already cached answer, never the network. /authz/check uses it
// for users-api itself: users-api answers GET /profile by calling /authz/check, so resolving
// there would loop.
func cachedPrincipal(subject string) models.Principal {
        return models.Principal{Subject: subject, UserID: userResolver.Cached(subject)}
}

// targetPrincipal is another user's login plus their canonical user, resolved with the acting
// administrator's bearer token. Without one (internal-token callers) it is the subject alone.
func targetPrincipal(ctx context.Context, subject, adminToken string) models.Principal {
        return models.Principal{Subject: subject, UserID: userResolver.Other(ctx, subject, adminToken)}
}

func buildSummary(ctx context.Context, subject, adminToken string) (subjectRolesSummary, error) {
        roles, err := models.ListRolesFor(ctx, targetPrincipal(ctx, subject, adminToken))
        if err != nil {
                return subjectRolesSummary{}, err
        }
        denials, err := models.ListDenyEntriesFor(ctx, targetPrincipal(ctx, subject, adminToken))
        if err != nil {
                return subjectRolesSummary{}, err
        }

        roleNames := make([]string, 0, len(roles))
        for _, r := range roles {
                roleNames = append(roleNames, string(r.Role))
        }
        if len(roleNames) == 0 {
                roleNames = []string{string(models.RoleUser)}
        }

        deniedServices := make([]string, 0, len(denials))
        for _, d := range denials {
                deniedServices = append(deniedServices, d.Service)
        }

        return subjectRolesSummary{Subject: subject, Roles: roleNames, DeniedServices: deniedServices}, nil
}

// Add a role to a subject.
//
//         Idempotent - a no-op, not a duplicate, if the subject already holds the role. Requires the internal service token, or an Auth0 bearer token with the admin role.
//                @Summary                Add a role to a subject
//                @Description        Add a role to a subject
//                @Tags                        admin
//                @Accept                        json
//                @Param                        X-Internal-Service-Token        header        string                                        false        "Shared internal-service secret"
//                @Param                        X-Acting-User-Sub                        header        string                                        false        "Acting admin's Auth0 sub, for audit attribution"
//                @Param                        subject                                                path        string                                        true        "Auth0 subject"
//                @Param                        request                                                body        roleAssignmentRequest        true        "Role to add"
//                @Success                201                                                        "Created"
//                @Success                204                                                        "No Content - role already present"
//                @Failure                400                                                        {object}        apiv.ErrorVO
//                @Failure                401                                                        {object}        apiv.ErrorVO
//                @Failure                403                                                        {object}        apiv.ErrorVO
//                @Failure                500                                                        {object}        apiv.ErrorVO
//                @Router                        /api/admin/roles/{subject} [post]
func addRole(c *gin.Context, cache *auth0.JWKSCache, config auth0.Config, authzClient *authz.Client, usersBaseURL string) {
        actingUserSub, token, ok := verifyAdminRole(c, cache, config)
        if !ok {
                return
        }

        actingUserID, ok := actingUserForAudit(c.Request.Context(), authzClient, usersBaseURL, actingUserSub, token)
        if !ok {
                c.JSON(http.StatusServiceUnavailable, apiv.ErrorVO{Error: "user_resolution_unavailable", Message: "unable to resolve the calling user"})
                return
        }

        subject := c.Param("subject")
        p := targetPrincipal(c.Request.Context(), subject, token)
        if subject == "" {
                c.JSON(http.StatusBadRequest, apiv.ErrorVO{Error: "invalid_request", Message: "invalid subject"})
                return
        }
        var req roleAssignmentRequest
        if err := c.ShouldBindJSON(&req); err != nil {
                c.JSON(http.StatusBadRequest, apiv.ErrorVO{Error: "invalid_request", Message: err.Error()})
                return
        }
        role, ok := models.ParseRole(req.Role)
        if !ok {
                c.JSON(http.StatusBadRequest, apiv.ErrorVO{Error: "invalid_request", Message: "invalid role"})
                return
        }

        status, err := audited(c.Request.Context(), actingUserID, "add_role", subject, string(role),
                func(ctx context.Context) (int, error) {
                        exists, err := hasRole(ctx, p, role)
                        if err != nil {
                                return 0, err
                        }
                        if exists {
                                return http.StatusNoContent, nil
                        }
                        if err := insertRole(ctx, p, role, actingUserID); err != nil {
                                return 0, err
                        }
                        return http.StatusCreated, nil
                })
        if err != nil {
                logging.Logger.Error("Failed to add role", "error", err.Error())
                c.JSON(http.StatusInternalServerError, apiv.ErrorVO{Error: "audit_failed", Message: "failed to add role"})
                return
        }
        c.Status(status)
}

// Remove a role from a subject.
//
//         Requires the internal service token, or an Auth0 bearer token with the admin role.
//                @Summary                Remove a role from a subject
//                @Description        Remove a role from a subject
//                @Tags                        admin
//                @Param                        X-Internal-Service-Token        header        string        false        "Shared internal-service secret"
//                @Param                        X-Acting-User-Sub                        header        string        false        "Acting admin's Auth0 sub, for audit attribution"

                func(ctx context.Context) (int, error) {

                return
        }
        if held {
                eventEmitter.Emit(events.KindRoleRemoved, subject, events.EventData{Role: string(role)})
        }
        c.Status(status)
}

// performAudited writes an AdminActionAuditLog row *before* running
// operation, and fails closed - never calling operation at all - if that
// write itself fails: an admin action that can't be logged must not be
// performed. Updates the same row to succeeded/failed after operation
// completes; that post-write is best-effort (logged on failure, but doesn't
// retroactively undo an action that already happened - the pre-write is the
// hard gate, not the post-write).
func performAudited(
        ctx context.Context, actingUserSub, action, targetSubject, detail string,
        operation func(context.Context) (int, error),
) (int, error) {
        return performAuditedWith(
                ctx, actingUserSub, action, targetSubject, detail, operation,
                models.RecordAuditAttempt, models.CompleteAudit,
        )
}

// performAuditedWith takes the audit read/write functions as parameters so
// the fail-closed contract above (operation must never run if the "before"
// audit write fails) can be verified in a test without a live database -
// see roles_audit_test.go.
func performAuditedWith(
        ctx context.Context, actingUserSub, action, targetSubject, detail string,
        operation func(context.Context) (int, error),
        recordAttempt func(context.Context, string, string, string, string) (uuid.UUID, error),
        complete func(context.Context, uuid.UUID, models.AuditStatus, string) error,
) (int, error) {
        auditID, err := recordAttempt(ctx, actingUserSub, action, targetSubject, detail)
        if err != nil {
                logging.Logger.Error("Refusing to perform admin action - audit log write failed", "action", action, "error", err.Error())
                return 0, fmt.Errorf("could not record audit log; action not performed: %w", err)
        }

        status, opErr := operation(ctx)
        if opErr != nil {
                if err := complete(ctx, auditID, models.AuditFailed, opErr.Error()); err != nil {
                        logging.Logger.Warn("Failed to record failure for audit log", "auditID", auditID, "error", err.Error())
                }
                return 0, opErr
        }

        if err := complete(ctx, auditID, models.AuditSucceeded, ""); err != nil {
                logging.Logger.Warn("Failed to record success for audit log", "auditID", auditID, "error", err.Error())
        }
        return status, nil
}

// verifyToken verifies a bearer token and returns its Auth0 subject. A variable so tests can
// stand in for Auth0's JWKS.
var verifyToken = func(ctx context.Context, cache *auth0.JWKSCache, token string, config auth0.Config) (string, error) {
        claims, err := cache.Verify(ctx, token, config)
        if err != nil {
                return "", err
        }
        return claims.Subject, nil
}

// holdsAdminRole reports whether the principal has the admin role. A variable so tests need no
// database.
var holdsAdminRole = func(ctx context.Context, p models.Principal) (bool, error) {
        roles, err := models.ListRolesFor(ctx, p)
        if err != nil {
                return false, err
        }
        for _, r := range roles {
                if r.Role == models.RoleAdmin {
                        return true, nil
                }
        }
        return false, nil
}

// verifyAdminRole resolves the acting administrator's subject and bearer token, or writes an
// error response and returns ok=false. The only credential is an Auth0 bearer token whose subject
// holds the admin role (PADR-0037: no service tokens); the token's own subject is the acting user.
func verifyAdminRole(c *gin.Context, cache *auth0.JWKSCache, config auth0.Config) (string, string, bool) {
        token := bearerToken(c)
        if token == "" {
                c.JSON(http.StatusUnauthorized, apiv.ErrorVO{Error: "unauthorized", Message: "missing or invalid credentials"})
                return "", "", false
        }

        subject, err := verifyToken(c.Request.Context(), cache, token, config)
        if err != nil {
                c.JSON(http.StatusUnauthorized, apiv.ErrorVO{Error: "unauthorized", Message: "missing or invalid credentials"})
                return "", "", false
        }

        isAdmin, err := holdsAdminRole(c.Request.Context(), ownPrincipal(c.Request.Context(), subject, token))
        if err != nil {
                logging.Logger.Error("Failed to look up roles", "error", err.Error())
                c.JSON(http.StatusInternalServerError, apiv.ErrorVO{Error: "query_failed", Message: "failed to verify admin role"})
                return "", "", false
        }
        if !isAdmin {
                c.JSON(http.StatusForbidden, apiv.ErrorVO{Error: "forbidden", Message: "admin role required"})
                return "", "", false
        }
        return subject, token, true
}

// actingUserForAudit returns who to attribute an admin write to. Attribution must not block the
// action, so when the canonical ID cannot be resolved (users-api unreachable, not configured, or
// refusing the lookup) the cause is logged at ERROR and the acting Auth0 subject is recorded
// instead. ok is false only when there is nothing at all to attribute to.
func actingUserForAudit(ctx context.Context, authzClient *authz.Client, usersBaseURL, actingUserSub, token string) (string, bool) {
        id, err := resolveActingUser(ctx, authzClient, usersBaseURL, actingUserSub, token)
        if err == nil {
                return id, true
        }
        logging.Logger.Error("could not resolve the acting user's canonical id; recording the Auth0 subject in the audit trail", "subject", actingUserSub, "error", err.Error())
        if actingUserSub == "" {
                return "", false
        }
        return actingUserSub, true
}

// resolveActingUser resolves the acting user's Auth0 subject to their canonical
// users._id with the caller's own bearer token (authzClient.ResolveUserID). Returns an error when
// there is no token or it resolves to nothing; actingUserForAudit then records the subject.
func resolveActingUser(ctx context.Context, authzClient *authz.Client, usersBaseURL, actingUserSub, token string) (string, error) {
        if token != "" {
                userID := authzClient.ResolveUserID(ctx, token)
                if userID == "" {
                        return "", fmt.Errorf("resolveActingUser: bearer token resolved to empty user ID")
                }
                return userID, nil
        }

        // An internal-service-token caller has no user bearer token to resolve with.
        return "", fmt.Errorf("resolveActingUser: no bearer token to resolve %q with", actingUserSub)
}

// SessionLimits are the abuse limits for the session endpoints. Defaults are PADR-0040's.
type SessionLimits struct {
        MaxOpenMints       int
        MaxSessionTuples   int
        MintsPerUserHour   int
        MintsGlobalHour    int
        TupleReqsPerMinute int
}

// SessionLimitsFromEnv reads the limits, falling back to the PADR-0040 defaults for any unset or
// non-positive value.
func SessionLimitsFromEnv() SessionLimits {
        num := func(name string, def int) int {
                if n, err := strconv.Atoi(util.GetEnv(name, "")); err == nil && n > 0 {
                        return n
                }
                return def
        }
        return SessionLimits{
                MaxOpenMints:       num("MAX_OPEN_MINTS", 100),
                MaxSessionTuples:   num("MAX_SESSION_TUPLES", 200),
                MintsPerUserHour:   num("MINTS_PER_USER_HOUR", 20),
                MintsGlobalHour:    num("MINTS_GLOBAL_HOUR", 600),
                TupleReqsPerMinute: num("SESSION_TUPLE_REQS_PER_MINUTE", 120),
        }
}

// SessionTiers are the rate-limit tiers the session endpoints use, for ratelimit.New.
func SessionTiers(l SessionLimits) map[string]ratelimit.Tier {
        return map[string]ratelimit.Tier{
                tierMintUser:    {Limit: l.MintsPerUserHour, Window: 3600},
                tierMintGlobal:  {Limit: l.MintsGlobalHour, Window: 3600},
                tierSessionTups: {Limit: l.TupleReqsPerMinute, Window: 60},
        }
}

type sessionCaller struct{ Subject, UserID string }

// authenticate verifies the bearer, resolves the canonical user, and refuses a user with a deny
// entry for Initiative! (PADR-0039). It writes the error response and returns ok=false on failure.
func (h sessionHandlers) authenticate(c *gin.Context) (sessionCaller, bool) {
        token := bearerToken(c)
        if token == "" {
                invalidTokenResponse(c)
                return sessionCaller{}, false
        }
        subject, err := verifyToken(c.Request.Context(), h.cache, token, h.config)
        if err != nil {
                invalidTokenResponse(c)
                return sessionCaller{}, false
        }
        p := principalFor(c.Request.Context(), subject, token)
        if p.UserID == "" {
                logging.Logger.Warn("Session request refused; canonical user unresolved", "subject", subject)
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return sessionCaller{}, false
        }
        denied, err := hasDenyEntry(c.Request.Context(), p, constants.InitiativeService)
        if err != nil {
                logging.Logger.Error("Failed to look up deny entry", "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return sessionCaller{}, false
        }
        if denied {
                c.JSON(http.StatusForbidden, authzErrorResponse{Error: "service_denied"})
                return sessionCaller{}, false
        }
        return sessionCaller{Subject: subject, UserID: p.UserID}, true
}

// allow spends one count of tier for key. No limiter, a counter-store error, or a spent budget all
// stop the request; the first two are 503, never unlimited.
func (h sessionHandlers) allow(c *gin.Context, tier, key string) bool {
        if h.limiter == nil {
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return false
        }
        res, err := h.limiter.Check(c.Request.Context(), key, tier)
        if err != nil {
                logging.Logger.Error("Session rate limiter unavailable; failing closed", "tier", tier, "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return false
        }
        if !res.Allowed {
                c.Header("Retry-After", strconv.Itoa(int(res.RetryAfter.Seconds())))
                c.JSON(http.StatusTooManyRequests, authzErrorResponse{Error: "rate_limited"})
                return false
        }
        return true
}

func userTuple(userID string) string { return userPrincipalType + userID }

// mint creates a session ID and the caller's GM tuple. The ID comes from here, never from a client.
func (h sessionHandlers) mint(c *gin.Context) {
        caller, ok := h.authenticate(c)
        if !ok || !h.allow(c, tierMintUser, caller.UserID) || !h.allow(c, tierMintGlobal, "all") {
                return
        }
        ctx := c.Request.Context()
        open, err := h.mints.CountOpen(ctx, caller.UserID)
        if err != nil {
                logging.Logger.Error("Mint count failed", "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        if open >= int64(h.limits.MaxOpenMints) {
                c.JSON(http.StatusTooManyRequests, authzErrorResponse{Error: "mint_cap"})
                return
        }
        id, err := newSessionID()
        if err != nil {
                c.JSON(http.StatusInternalServerError, authzErrorResponse{Error: "query_failed"})
                return
        }
        gm := fga.Tuple{User: userTuple(caller.UserID), Relation: relationGM, Object: sessionObjectPrefix + id}
        now := nowFunc()
        mint := models.SessionMint{ID: id, UserID: caller.UserID, CreatedAt: now, CreatedBy: caller.UserID, UpdatedAt: now, UpdatedBy: caller.UserID}
        _, err = auditedWrite(ctx, caller.Subject, actionMint, gm.Object, "user="+caller.UserID+"; "+auditDetail(tupleWriteRequest{Writes: []fga.Tuple{gm}}),
                func(ctx context.Context) (int, error) {
                        if err := h.mints.Insert(ctx, mint); err != nil {
                                return 0, err
                        }
                        if err := h.store.Write(ctx, []fga.Tuple{gm}, nil); err != nil {
                                if _, cerr := h.mints.Close(ctx, id, caller.UserID, nowFunc()); cerr != nil {
                                        logging.Logger.Warn("Could not close mint after a failed tuple write", "mint", id, "error", cerr.Error())
                                }
                                return 0, err
                        }
                        return http.StatusCreated, nil
                })
        if err != nil {
                logging.Logger.Error("Session mint failed", "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        c.JSON(http.StatusCreated, gin.H{"session_id": id})
}

// end deletes every tuple on a session and closes its mint. Only the GM may.
func (h sessionHandlers) end(c *gin.Context) {
        caller, ok := h.authenticate(c)
        if !ok {
                return
        }
        id := c.Param("id")
        if !sessionIDPattern.MatchString(id) {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        if !h.allow(c, tierSessionTups, caller.UserID) {
                return
        }
        object := sessionObjectPrefix + id
        allowed, err := h.store.Check(c.Request.Context(), fga.Tuple{User: userTuple(caller.UserID), Relation: relationGM, Object: object})
        if err != nil {
                logging.Logger.Error("OpenFGA check failed", "object", object, "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        if !allowed && !h.ownsEmptyOpenMint(c.Request.Context(), caller.UserID, id, object) {
                c.JSON(http.StatusForbidden, authzErrorResponse{Error: "forbidden"})
                return
        }
        if err := h.removeSession(c.Request.Context(), caller.Subject, caller.UserID, actionEnd, id); err != nil {
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        c.Status(http.StatusNoContent)
}

// ownsEmptyOpenMint reports whether the session has no tuples left and the caller has its mint
// still open: a retry of an end whose tuples were deleted but whose mint close failed. Closing it
// again is safe because nothing is left to grant. Any error or other state answers false (403).
func (h sessionHandlers) ownsEmptyOpenMint(ctx context.Context, userID, id, object string) bool {
        left, _, err := h.store.Read(ctx, object, 1, "")
        if err != nil || len(left) > 0 {
                return false
        }
        mint, open, err := h.mints.FindOpen(ctx, id)
        return err == nil && open && mint.UserID == userID
}

// removeSession is the audited removal of a session's tuples and its mint, shared by the GM's end
// and the reconciler's garbage-collection close.
func (h sessionHandlers) removeSession(ctx context.Context, subject, actor, action, id string) error {
        object := sessionObjectPrefix + id
        _, err := auditedWrite(ctx, subject, action, object, "user="+actor+"; delete all tuples on "+object, func(ctx context.Context) (int, error) {
                if err := deleteAllTuples(ctx, h.store, object); err != nil {
                        return 0, err
                }
                if _, err := h.mints.Close(ctx, id, actor, nowFunc()); err != nil {
                        return 0, err
                }
                return http.StatusOK, nil
        })
        if err != nil {
                logging.Logger.Error("Session removal failed", "object", object, "error", err.Error())
        }
        return err
}

// deleteAllTuples removes every tuple on object, a page at a time. Deleting a missing tuple is
// ignored by the store, so a retry after a partial run is safe.
func deleteAllTuples(ctx context.Context, store fga.Store, object string) error {
        for range maxDeleteRounds {
                page, _, err := store.Read(ctx, object, maxTuplePage, "")
                if err != nil {
                        return err
                }
                if len(page) == 0 {
                        return nil
                }
                if err := store.Write(ctx, nil, page); err != nil {
                        return err
                }
        }
        return errors.New("session still has tuples after the delete rounds")
}

// readTuples returns the tuples on object, stopping once it has more than limit (enough to know
// the session is full).
func readTuples(ctx context.Context, store fga.Store, object string, limit int) ([]fga.Tuple, error) {
        var all []fga.Tuple
        token := ""
        for len(all) <= limit {
                page, next, err := store.Read(ctx, object, maxTuplePage, token)
                if err != nil {
                        return nil, err
                }
                all = append(all, page...)
                if next == "" {
                        break
                }
                token = next
        }
        return all, nil
}

// netTuples is how many tuples the session has after the change: the current set plus the writes
// not already present, minus the deletes that are present. OpenFGA ignores a duplicate write and a
// missing delete, so counting them would let a caller pad deletes to slip writes past the cap.
func netTuples(current, writes, deletes []fga.Tuple) int {
        have := make(map[fga.Tuple]bool, len(current))
        for _, t := range current {
                have[t] = true
        }
        n := len(current)
        for _, t := range writes {
                if !have[t] {
                        have[t] = true
                        n++
                }
        }
        for _, t := range deletes {
                if have[t] {
                        delete(have, t)
                        n--
                }
        }
        return n
}

type sessionTuple struct {
        User     string `json:"user"`
        Relation string `json:"relation"`
}

type sessionTuplesRequest struct {
        SessionID string         `json:"session_id"`
        Writes    []sessionTuple `json:"writes"`
        Deletes   []sessionTuple `json:"deletes"`
}

// requiredRelation is what the caller must hold on the session for this change (PADR-0040 rule
// table): can_view to delete only their own tuple (leaving), gm for any co_gm tuple, can_edit
// otherwise. ok is false when a tuple is malformed or uses a relation this endpoint never writes.
func requiredRelation(req sessionTuplesRequest, callerUser string) (rel string, ok bool) {
        n := len(req.Writes) + len(req.Deletes)
        if n == 0 || n > maxSessionTupleBatch {
                return "", false
        }
        needGM, ownOnly := false, len(req.Writes) == 0
        for _, t := range append(append([]sessionTuple{}, req.Writes...), req.Deletes...) {
                if !validUser(t.User) || !memberRelations[t.Relation] {
                        return "", false
                }
                needGM = needGM || t.Relation == relationCoGM
                ownOnly = ownOnly && t.User == userTuple(callerUser)
        }
        switch {
        case ownOnly:
                return relCanView, true
        case needGM:
                return relationGM, true
        }
        return relCanEdit, true
}

func objectTuples(object string, ts []sessionTuple) []fga.Tuple {
        out := make([]fga.Tuple, 0, len(ts))
        for _, t := range ts {
                out = append(out, fga.Tuple{User: t.User, Relation: t.Relation, Object: object})
        }
        return out
}

// writeSessionTuples applies membership tuple changes on one session after checking the caller's
// own relation on it. The object is always session:<id>, so no other object type is reachable.
// The tuples are necessary but not sufficient: initiative-api still requires a live Mongo
// membership record, and its reconciler removes tuples Mongo does not back.
func (h sessionHandlers) writeSessionTuples(c *gin.Context) {
        caller, ok := h.authenticate(c)
        if !ok {
                return
        }
        var req sessionTuplesRequest
        if err := c.ShouldBindJSON(&req); err != nil || !sessionIDPattern.MatchString(req.SessionID) {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        need, valid := requiredRelation(req, caller.UserID)
        if !valid {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        if !h.allow(c, tierSessionTups, caller.UserID) {
                return
        }
        ctx, object := c.Request.Context(), sessionObjectPrefix+req.SessionID
        held, err := h.store.Check(ctx, fga.Tuple{User: userTuple(caller.UserID), Relation: need, Object: object})
        if err != nil {
                logging.Logger.Error("OpenFGA check failed", "object", object, "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        if !held {
                c.JSON(http.StatusForbidden, authzErrorResponse{Error: "forbidden"})
                return
        }
        writes, deletes := objectTuples(object, req.Writes), objectTuples(object, req.Deletes)
        if len(writes) > 0 {
                current, err := readTuples(ctx, h.store, object, h.limits.MaxSessionTuples+maxSessionTupleBatch)
                if err != nil {
                        c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                        return
                }
                if netTuples(current, writes, deletes) > h.limits.MaxSessionTuples {
                        c.JSON(http.StatusConflict, authzErrorResponse{Error: "session_full"})
                        return
                }
        }
        detail := fmt.Sprintf("user=%s; %s", caller.UserID, auditDetail(tupleWriteRequest{Writes: writes, Deletes: deletes}))
        if _, err := auditedWrite(ctx, caller.Subject, actionTuples, object, detail, func(ctx context.Context) (int, error) {
                return http.StatusOK, h.store.Write(ctx, writes, deletes)
        }); err != nil {
                logging.Logger.Error("Session tuple write failed", "object", object, "error", err.Error())
                if errors.Is(err, fga.ErrInvalid) {
                        c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_tuples"})
                        return
                }
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        c.JSON(http.StatusOK, gin.H{"writes": len(writes), "deletes": len(deletes)})
}

// listMints returns open mints at least min_age old by auth-api's own clock, for the reconciler's
// garbage collection, ordered by (created_at, id) and paged: `after` is the `next` value of the
// previous page, and `next` is set only when more remain. It requires openfga-writer.
func (h sessionHandlers) listMints(c *gin.Context) {
        if _, ok := h.requireWriterStatus(c, http.StatusServiceUnavailable, authzUnavailable); !ok {
                return
        }
        age, err := time.ParseDuration(c.Query("min_age"))
        if err != nil || age < 0 || age > maxMintMinAge {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        limit := int64(100)
        if raw := c.Query("limit"); raw != "" {
                n, err := strconv.Atoi(raw)
                if err != nil || n < 1 || n > maxMintList {
                        c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                        return
                }
                limit = int64(n)
        }
        after, ok := parseMintCursor(c.Query("after"))
        if !ok {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        mints, err := h.mints.ListOpenBefore(c.Request.Context(), nowFunc().Add(-age), after, limit+1)
        if err != nil {
                logging.Logger.Error("Mint list failed", "error", err.Error())
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        next := ""
        if int64(len(mints)) > limit {
                mints = mints[:limit]
                next = formatMintCursor(mints[len(mints)-1])
        }
        if mints == nil {
                mints = []models.SessionMint{}
        }
        c.JSON(http.StatusOK, gin.H{"mints": mints, "next": next})
}

// formatMintCursor encodes the position after m as "<unix nanos>.<id>".
func formatMintCursor(m models.SessionMint) string {
        return strconv.FormatInt(m.CreatedAt.UnixNano(), 10) + "." + m.ID
}

// parseMintCursor decodes a cursor; an empty one means the start. ok is false for a malformed one.
func parseMintCursor(raw string) (*models.MintCursor, bool) {
        if raw == "" {
                return nil, true
        }
        nanos, id, found := strings.Cut(raw, ".")
        n, err := strconv.ParseInt(nanos, 10, 64)
        if !found || err != nil || !sessionIDPattern.MatchString(id) {
                return nil, false
        }
        return &models.MintCursor{CreatedAt: time.Unix(0, n).UTC(), ID: id}, true
}

// closeMint deletes every tuple on a minted session and closes its mint. The reconciler calls it
// for a mint whose session Mongo does not have (or has deleted). It requires openfga-writer.
func (h sessionHandlers) closeMint(c *gin.Context) {
        subject, ok := h.requireWriterStatus(c, http.StatusServiceUnavailable, authzUnavailable)
        if !ok {
                return
        }
        id := c.Param("id")
        if !sessionIDPattern.MatchString(id) {
                c.JSON(http.StatusBadRequest, authzErrorResponse{Error: "invalid_request"})
                return
        }
        if err := h.removeSession(c.Request.Context(), subject, subject, actionMintClose, id); err != nil {
                c.JSON(http.StatusServiceUnavailable, authzErrorResponse{Error: authzUnavailable})
                return
        }
        c.Status(http.StatusNoContent)
}
